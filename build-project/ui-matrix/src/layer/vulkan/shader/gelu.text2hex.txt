#version 450
#define shape_constant_id_offset 0
layout(constant_id = 0) const uint n = 0;
layout(binding = 0) buffer bottom_top_blob { sfpvec4 bottom_top_blob_data[]; };
layout(push_constant) uniform parameter
{
uint n;
} p;
void main()
{
uint gi = gl_GlobalInvocationID.x;
if (gi >= psc(n))
return;
afpvec4 v = buffer_ld4(bottom_top_blob_data, gi);
afpvec4 x = afpvec4(0.79788452f) * (v + afpvec4(0.044715f) * v * v * v);
x = clamp(x, afpvec4(-10.f), afpvec4(10.f));
#if NCNN_moltenvk
v = afpvec4(0.5f) * v * (afpvec4(1.0f) + afpvec4(tanh(vec4(x))));
#else
v = afpvec4(0.5f) * v * (afpvec4(1.0f) + tanh(x));
#endif
buffer_st4(bottom_top_blob_data, gi, v);
}
