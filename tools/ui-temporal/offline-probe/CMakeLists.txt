cmake_minimum_required(VERSION 3.24)
project(OfflineTemporalUiProbe LANGUAGES CXX)
if(NOT WIN32 OR NOT MSVC)
    message(FATAL_ERROR "This standalone diagnostic requires Windows/MSVC and existing DirectML")
endif()
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
add_executable(offline_temporal_probe "${CMAKE_CURRENT_LIST_DIR}/../offline_temporal_probe.cpp")
target_include_directories(offline_temporal_probe PRIVATE "${ROOT}/third_party/ai-depth/include")
target_compile_features(offline_temporal_probe PRIVATE cxx_std_17)
target_compile_options(offline_temporal_probe PRIVATE /W4 /EHsc)
target_link_libraries(offline_temporal_probe PRIVATE d3d12 dxgi)
